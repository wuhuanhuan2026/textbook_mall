import axios from 'axios'
import { getToken, removeToken, removeUser } from '@/utils/storage'
import { ElMessage } from 'element-plus'

/**
 * Axios 实例配置
 * baseURL: 从环境变量读取，默认 /api
 * timeout: 10秒超时
 */
const request = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || '/api',
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json'
  }
})

/**
 * 请求拦截器
 * 在请求发送前自动添加 Token
 */
request.interceptors.request.use(
  (config) => {
    const token = getToken()
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (error) => Promise.reject(error)
)

/**
 * 响应拦截器
 * 统一处理响应数据、错误提示、Token 过期
 */
request.interceptors.response.use(
  (response) => {
    const { code, message, data } = response.data
    
    // 成功响应
    if (code === 200) {
      return data
    }
    
    // Token 过期或未登录 (401 / 1010)
    if (code === 401 || code === 1010) {
      ElMessage.error('登录已过期，请重新登录')
      removeToken()
      removeUser()
      // 跳转到登录页
      if (window.location.pathname !== '/login') {
        window.location.href = '/login'
      }
      return Promise.reject(new Error('Unauthorized'))
    }
    
    // 业务错误
    ElMessage.error(message || '请求失败')
    return Promise.reject(new Error(message))
  },
  (error) => {
    // 网络错误
    ElMessage.error(error.message || '网络错误，请稍后重试')
    return Promise.reject(error)
  }
)

export default request

  book: {
    type: Object,
    required: true,
    validator: (value) => {
      return value && typeof value.bookId !== 'undefined'
    }
  }
})

  if (!props.book.originalPrice || props.book.originalPrice === 0) return 0
  return Math.round((props.book.price / props.book.originalPrice) * 10)
})
</script>

import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { cartApi } from '@/api/cart'

/**
 * 购物车状态管理
 */
export const useCartStore = defineStore('cart', () => {
  // ========== State ==========
  const cartItems = ref([])
  const loading = ref(false)

  // ========== Getters ==========
  const totalCount = computed(() => {
    return cartItems.value.reduce((sum, item) => sum + item.quantity, 0)
  })

  const totalPrice = computed(() => {
    return cartItems.value.reduce((sum, item) => sum + (item.price || 0) * item.quantity, 0)
  })

  // ========== Actions ==========
  /**
   * 获取购物车列表
   */
  const fetchCart = async () => {
    loading.value = true
    try {
      const items = await cartApi.getCartList()
      cartItems.value = items
    } finally {
      loading.value = false
    }
  }

  /**
   * 添加到购物车
   */
  const addToCart = async (bookId, quantity = 1) => {
    await cartApi.addToCart({ bookId, quantity })
    await fetchCart()
  }

  /**
   * 更新商品数量
   */
  const updateQuantity = async (cartId, quantity) => {
    await cartApi.updateCartItem(cartId, { quantity })
    await fetchCart()
  }

  /**
   * 删除购物车项
   */
  const removeItem = async (cartId) => {
    await cartApi.deleteCartItem(cartId)
    await fetchCart()
  }

  /**
   * 清空购物车
   */
  const clearCart = async () => {
    await cartApi.clearCart()
    cartItems.value = []
  }

  return {
    cartItems,
    loading,
    totalCount,
    totalPrice,
    fetchCart,
    addToCart,
    updateQuantity,
    removeItem,
    clearCart
  }
})

import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { getToken, setToken, removeToken, getUser, setUser, removeUser } from '@/utils/storage'
import { userApi } from '@/api/user'

/**
 * 用户状态管理
 */
export const useUserStore = defineStore('user', () => {
  // ========== State ==========
  const token = ref(getToken() || '')
  const userInfo = ref(getUser())
  
  // ========== Getters ==========
  const isLoggedIn = computed(() => !!token.value)

  // ========== Actions ==========
  /**
   * 设置认证信息
   */
  const setAuth = (data) => {
    token.value = data.token
    setToken(data.token)
    userInfo.value = {
      userId: data.userId,
      username: data.username,
      avatar: data.avatar || '',
      phone: '',
      email: '',
      balance: data.balance,
      status: 1,
      createTime: ''
    }
    setUser(userInfo.value)
  }

  /**
   * 清除认证信息
   */
  const clearAuth = () => {
    token.value = ''
    userInfo.value = null
    removeToken()
    removeUser()
  }

  /**
   * 登录
   */
  const login = async (params) => {
    const result = await userApi.login(params)
    setAuth(result)
    return result
  }

  /**
   * 退出登录
   */
  const logout = async () => {
    try {
      await userApi.logout()
    } finally {
      clearAuth()
    }
  }

  /**
   * 获取用户信息
   */
  const fetchUserInfo = async () => {
    const info = await userApi.getUserInfo()
    userInfo.value = info
    setUser(info)
    return info
  }

  /**
   * 更新用户信息
   */
  const updateUserInfo = async (params) => {
    // TODO: 调用 API
    // await userApi.updateUserInfo(params)
    if (userInfo.value) {
      userInfo.value = { ...userInfo.value, ...params }
      setUser(userInfo.value)
    }
  }

  return {
    // State
    token,
    userInfo,
    // Getters
    isLoggedIn,
    // Actions
    login,
    logout,
    fetchUserInfo,
    updateUserInfo,
    clearAuth
  }
})
