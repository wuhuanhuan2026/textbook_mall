// frontend/src/api/request.js
import axios from 'axios'
import { getToken, removeToken, removeUser } from '@/utils/storage'
import { ElMessage } from 'element-plus'

/**
 * Axios 实例配置
 * baseURL: /api，与 Gateway 路由匹配
 * Gateway 的 StripPrefix=1 会移除 /api 前缀，转发到后端服务
 */
const request = axios.create({
  baseURL: '/api',  // ✅ 改为 /api，与 Gateway 路由匹配
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json'
  }
})

/**
 * 请求拦截器
 * 在请求发送前自动添加 Token
 */
request.interceptors.request.use(
  (config) => {
    const token = getToken()
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (error) => Promise.reject(error)
)

/**
 * 响应拦截器
 * 统一处理响应数据、错误提示、Token 过期
 */
request.interceptors.response.use(
  (response) => {
    const { code, message, data } = response.data
    
    // 成功响应
    if (code === 200) {
      return response.data
    }
    
    // Token 过期或未登录 (401 / 1010)
    if (code === 401 || code === 1010) {
      ElMessage.error('登录已过期，请重新登录')
      removeToken()
      removeUser()
      // 跳转到登录页
      if (window.location.pathname !== '/login' && window.location.pathname !== '/admin/login') {
        window.location.href = '/login'
      }
      return Promise.reject(new Error('Unauthorized'))
    }
    
    // 业务错误
    ElMessage.error(message || '请求失败')
    return Promise.reject(new Error(message))
  },
  (error) => {
    // HTTP 错误
    const status = error.response?.status
    const message = error.response?.data?.message || error.message || '网络错误，请稍后重试'
    
    if (status === 401) {
      ElMessage.error('登录已过期，请重新登录')
      removeToken()
      removeUser()
      if (window.location.pathname !== '/login' && window.location.pathname !== '/admin/login') {
        window.location.href = '/login'
      }
    } else if (status === 403) {
      ElMessage.error('权限不足，请联系管理员')
    } else if (status === 404) {
      ElMessage.error('请求的资源不存在')
    } else if (status === 500) {
      ElMessage.error('服务器内部错误，请稍后重试')
    } else {
      ElMessage.error(message)
    }
    
    return Promise.reject(error)
  }
)

export default request

/**
 * ✅ 安全解码 JWT payload（支持 URL-safe Base64）
 */
function decodeJwtPayload(token) {
  if (!token) return null
  
  try {
    const parts = token.split('.')
    if (parts.length !== 3) {
      console.error('❌ JWT 格式无效，段数:', parts.length)
      return null
    }
    
    // 获取 payload 部分
    let base64 = parts[1]
    
    // 1. 替换 URL-safe 字符为标准 Base64 字符
    base64 = base64.replace(/-/g, '+').replace(/_/g, '/')
    
    // 2. 补齐 padding（Base64 要求长度为 4 的倍数）
    while (base64.length % 4 !== 0) {
      base64 += '='
    }
    
    // 3. 解码
    const jsonStr = atob(base64)
    return JSON.parse(jsonStr)
  } catch (error) {
    console.error('❌ JWT 解码失败:', error.message)
    return null
  }
}

import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { cartApi } from '@/api/cart'
import { ElMessage } from 'element-plus'

/**
 * 购物车状态管理
 */
export const useCartStore = defineStore('cart', () => {
  // ========== State ==========
  const cartItems = ref([])
  const loading = ref(false)

  // ========== Getters ==========
  const totalCount = computed(() => {
    if (!cartItems.value || !Array.isArray(cartItems.value)) {
      return 0
    }
    return cartItems.value.reduce((sum, item) => sum + (item.quantity || 0), 0)
  })

  const totalPrice = computed(() => {
    if (!cartItems.value || !Array.isArray(cartItems.value)) {
      return 0
    }
    return cartItems.value.reduce((sum, item) => {
      const price = Number(item.price) || 0
      const quantity = Number(item.quantity) || 0
      return sum + price * quantity
    }, 0)
  })

  // ========== Actions ==========
  /**
   * 获取购物车列表
   */
  const fetchCart = async () => {
    loading.value = true
    try {
      const res = await cartApi.getCartList()
      if (res && res.code === 200) {
        cartItems.value = Array.isArray(res.data) ? res.data : []
      } else if (Array.isArray(res)) {
        cartItems.value = res
      } else {
        cartItems.value = []
      }
    } catch (error) {
      console.error('获取购物车失败:', error)
      cartItems.value = []
    } finally {
      loading.value = false
    }
  }

  /**
   * 添加到购物车
   * ✅ 修复：传递对象 { bookId, quantity }
   */
  const addToCart = async (bookId, quantity = 1) => {
    try {
      // ✅ 关键修复：传递对象，而不是两个独立参数
      const res = await cartApi.addToCart({ bookId, quantity })
      console.log('addToCart 响应:', res)
      
      if (res && res.code === 200) {
        await fetchCart()
        ElMessage.success('已加入购物车')
        return res.data
      }
      throw new Error(res?.message || '添加失败')
    } catch (error) {
      console.error('加入购物车失败:', error)
      ElMessage.error(error.message || '添加失败')
      throw error
    }
  }

  /**
   * 更新商品数量
   */
  const updateQuantity = async (cartId, quantity) => {
    try {
      const res = await cartApi.updateCartItem(cartId, { quantity })
      if (res && res.code === 200) {
        await fetchCart()
        return res.data
      }
      throw new Error(res?.message || '更新失败')
    } catch (error) {
      ElMessage.error(error.message || '更新失败')
      throw error
    }
  }

  /**
   * 删除购物车项
   */
  const removeItem = async (cartId) => {
    try {
      const res = await cartApi.deleteCartItem(cartId)
      if (res && res.code === 200) {
        await fetchCart()
        ElMessage.success('已删除')
        return true
      }
      throw new Error(res?.message || '删除失败')
    } catch (error) {
      ElMessage.error(error.message || '删除失败')
      throw error
    }
  }

  /**
   * 清空购物车
   */
  const clearCart = async () => {
    try {
      const res = await cartApi.clearCart()
      if (res && res.code === 200) {
        cartItems.value = []
        ElMessage.success('已清空购物车')
        return true
      }
      throw new Error(res?.message || '清空失败')
    } catch (error) {
      ElMessage.error(error.message || '清空失败')
      throw error
    }
  }

  return {
    cartItems,
    loading,
    totalCount,
    totalPrice,
    fetchCart,
    addToCart,
    updateQuantity,
    removeItem,
    clearCart
  }
})

import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { getToken, setToken, removeToken, getUser, setUser, removeUser } from '@/utils/storage'
import { userApi } from '@/api/user'

/**
 * 用户状态管理
 */
export const useUserStore = defineStore('user', () => {
  // ========== State ==========
  const token = ref(getToken() || '')
  const userInfo = ref(getUser())
  
  // ========== Getters ==========
  const isLoggedIn = computed(() => !!token.value)

  // ========== Actions ==========
  /**
   * 设置认证信息
   */
  const setAuth = (data) => {
    token.value = data.token
    setToken(data.token)
    userInfo.value = {
      userId: data.userId,
      username: data.username,
      avatar: data.avatar || '',
      phone: data.phone || '',
      email: data.email || '',
      balance: data.balance || 0,
      status: data.status || 1,
      createTime: data.createTime || ''
    }
    setUser(userInfo.value)
  }

  /**
   * 清除认证信息
   */
  const clearAuth = () => {
    token.value = ''
    userInfo.value = null
    removeToken()
    removeUser()
  }

  /**
   * 登录
   */
  const login = async (params) => {
    const res = await userApi.login(params)
    // 检查响应是否成功
    if (res.code === 200 && res.data) {
      setAuth(res.data)
      return res
    } else {
      throw new Error(res.message || '登录失败')
    }
  }

  /**
   * 退出登录
   */
  const logout = async () => {
    try {
      await userApi.logout()
    } catch (error) {
      console.error('退出请求失败:', error)
    } finally {
      clearAuth()
    }
  }

  /**
   * 获取用户信息
   */
  const fetchUserInfo = async () => {
    try {
      const res = await userApi.getUserInfo()
      if (res.code === 200 && res.data) {
        userInfo.value = res.data
        setUser(res.data)
        return res.data
      }
      return null
    } catch (error) {
      console.error('获取用户信息失败:', error)
      throw error
    }
  }

  /**
   * 更新用户信息
   */
  const updateUserInfo = async (params) => {
    try {
      const res = await userApi.updateUserInfo(params)
      if (res.code === 200 && res.data) {
        userInfo.value = { ...userInfo.value, ...res.data }
        setUser(userInfo.value)
        return res.data
      }
      throw new Error(res.message || '更新失败')
    } catch (error) {
      console.error('更新用户信息失败:', error)
      throw error
    }
  }

  return {
    // State
    token,
    userInfo,
    // Getters
    isLoggedIn,
    // Actions
    login,
    logout,
    fetchUserInfo,
    updateUserInfo,
    clearAuth
  }
})

import { ref, reactive, onMounted, computed } from 'vue'

// 密码强度计算
const passwordStrength = computed(() => {
  const pwd = pwdForm.newPassword
  if (!pwd || pwd.length < 6) return 'weak'
  let score = 0
  if (pwd.length >= 8) score++
  if (/[a-z]/.test(pwd) && /[A-Z]/.test(pwd)) score++
  if (/\d/.test(pwd)) score++
  if (/[^a-zA-Z0-9]/.test(pwd)) score++
  if (pwd.length >= 12) score++
  if (score >= 4) return 'strong'
  if (score >= 2) return 'medium'
  return 'weak'
})

const strengthText = computed(() => {
  const map = { weak: '弱', medium: '中等', strong: '强' }
  return map[passwordStrength.value] || ''
})

const rules = {

      :title="dialogTitle"
      width="480px"
      @close="resetForm"
    >
      <el-form ref="formRef" :model="form" :rules="rules" label-width="80px">
        <el-form-item label="用户名" prop="username">
          <el-input v-model="form.username" placeholder="请输入用户名" :disabled="isEdit" />
        </el-form-item>
        <el-form-item label="姓名" prop="realName">
          <el-input v-model="form.realName" placeholder="请输入真实姓名" />
        </el-form-item>
        <el-form-item label="角色" prop="role">
          <el-select v-model="form.role" placeholder="请选择角色" style="width:100%">
            <el-option label="超级管理员" value="super_admin" />
            <el-option label="管理员" value="admin" />
            <el-option label="运营人员" value="operator" />
            <el-option label="查看者" value="viewer" />
          </el-select>
        </el-form-item>
        <el-form-item label="手机号" prop="phone">
          <el-input v-model="form.phone" placeholder="请输入手机号" />
        </el-form-item>
        <el-form-item label="邮箱" prop="email">
          <el-input v-model="form.email" placeholder="请输入邮箱" />
        </el-form-item>
        <el-form-item label="头像URL" prop="avatar">
          <el-input v-model="form.avatar" placeholder="请输入头像图片URL" />
        </el-form-item>
        <el-form-item v-if="!isEdit" label="初始密码">
          <el-alert title="默认密码为 123456，请管理员登录后修改" type="info" :closable="false" />
        </el-form-item>
      </el-form>
      <template #footer>
        <el-button @click="dialogVisible = false">取消</el-button>
        <el-button type="primary" @click="submitForm" :loading="submitting">
          {{ isEdit ? '保存修改' : '确认添加' }}
        </el-button>
      </template>
    </el-dialog>
  </div>
</template>

<script setup>
import { ref, reactive, onMounted, computed } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { adminApi } from '@/api/admin'

const loading = ref(false)
const submitting = ref(false)
const adminList = ref([])
const dialogVisible = ref(false)
const isEdit = ref(false)
const formRef = ref()

const form = reactive({
  adminId: null,
  username: '',
  realName: '',
  role: 'admin',
  phone: '',
  email: '',
  avatar: ''
})

const dialogTitle = computed(() => isEdit.value ? '编辑管理员' : '新增管理员')

const rules = {
  username: [{ required: true, message: '请输入用户名', trigger: 'blur' }],
  realName: [{ required: true, message: '请输入真实姓名', trigger: 'blur' }],
  role: [{ required: true, message: '请选择角色', trigger: 'change' }]
}

const getRoleText = (role) => {
  const map = {
    super_admin: '超级管理员',
    admin: '管理员',
    operator: '运营人员',
    viewer: '查看者'
  }
  return map[role] || role
}

const fetchAdmins = async () => {
  loading.value = true
  try {
    const res = await adminApi.getAdminList()
    if (res && res.code === 200) {
      adminList.value = res.data || []
    }
  } catch (error) {
    console.error('获取管理员列表失败:', error)
  } finally {
    loading.value = false
  }
}

const showCreateDialog = () => {
  isEdit.value = false
  form.adminId = null
  resetForm()
  dialogVisible.value = true
}

const editAdmin = (row) => {
  isEdit.value = true
  form.adminId = row.adminId
  form.username = row.username
  form.realName = row.realName || ''
  form.role = row.role
  form.phone = row.phone || ''
  form.email = row.email || ''
  form.avatar = row.avatar || ''
  dialogVisible.value = true
}

const resetForm = () => {
  form.username = ''
  form.realName = ''
  form.role = 'admin'
  form.phone = ''
  form.email = ''
  form.avatar = ''
  if (formRef.value) {
    formRef.value.resetFields()
  }
}

const submitForm = async () => {
  if (!formRef.value) return
  
  await formRef.value.validate(async (valid) => {
    if (!valid) return
    
    submitting.value = true
    try {
      const data = {
        username: form.username,
        realName: form.realName,
        role: form.role,
        phone: form.phone,
        email: form.email,
        avatar: form.avatar
      }
      
      if (isEdit.value) {
        await adminApi.updateAdmin(form.adminId, data)
        ElMessage.success('更新成功')
      } else {
        await adminApi.createAdmin(data)
        ElMessage.success('添加成功，默认密码为 123456')
      }
      dialogVisible.value = false
      fetchAdmins()
    } catch (error) {
      ElMessage.error(isEdit.value ? '更新失败' : '添加失败')
    } finally {
      submitting.value = false
    }
  })
}

const handleEnable = async (adminId) => {
  try {
    await ElMessageBox.confirm('确定要启用该管理员吗？', '提示', { type: 'info' })
    await adminApi.enableAdmin(adminId)
    ElMessage.success('启用成功')
    fetchAdmins()
  } catch (error) {
    if (error !== 'cancel') {
      ElMessage.error('启用失败')
    }
  }
}

const handleDisable = async (adminId) => {
  try {
    await ElMessageBox.confirm('确定要禁用该管理员吗？', '提示', { type: 'warning' })
    await adminApi.disableAdmin(adminId)
    ElMessage.success('禁用成功')
    fetchAdmins()
  } catch (error) {
    if (error !== 'cancel') {
      ElMessage.error('禁用失败')
    }
  }
}

const handleDelete = async (adminId) => {
  try {
    await ElMessageBox.confirm('确定要删除该管理员吗？此操作不可恢复！', '危险操作', { type: 'error' })
    await adminApi.deleteAdmin(adminId)
    ElMessage.success('删除成功')
    fetchAdmins()
  } catch (error) {
    if (error !== 'cancel') {
      ElMessage.error('删除失败')
    }
  }
}

onMounted(fetchAdmins)
</script>

.admin-users {
  max-width: 1200px;
  margin: 0 auto;
}

  display: flex;
  justify-content: space-between;
  align-items: center;
}
</style>
